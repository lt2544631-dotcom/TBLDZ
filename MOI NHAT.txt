import random
import streamlit as st

st.title("🎮 Trò chơi Đoán Số")
st.write(
    "Tôi đã chọn một số từ **1 đến 100**. Bạn hãy đoán xem đó là số mấy nhé!"
)

# Khởi tạo trạng thái trò chơi
if "target_number" not in st.session_state:
  st.session_state.target_number = random.randint(1, 100)
  st.session_state.attempts = 0

# Nhập số từ người chơi
guess = st.number_input(
    "Nhập số của bạn:", min_value=1, max_value=100, step=1
)

if st.button("Đoán"):
  st.session_state.attempts += 1
  if guess < st.session_state.target_number:
    st.warning("Số bạn đoán hơi **nhỏ** rồi! Thử lại xem.")
  elif guess > st.session_state.target_number:
    st.warning("Số bạn đoán hơi **lớn** rồi! Thử lại xem.")
  else:
    st.success(
        f"🎉 Chính xác! Bạn đã đoán đúng số {st.session_state.target_number}"
        f" sau {st.session_state.attempts} lần thử."
    )
    if st.button("Chơi lại"):
      st.session_state.target_number = random.randint(1, 100)
      st.session_state.attempts = 0
      st.rerun()
